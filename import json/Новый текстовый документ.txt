CREATE TABLE RT2 (
       id int NOT NULL AUTO_INCREMENT,
       box varchar(150) NOT NULL,
       bigbox varchar(250) NOT NULL,
       pallet varchar(250) NOT NULL,
       time_stamp datetime NOT NULL DEFAULT CURRENT_TIMESTAMP,
       PRIMARY KEY (id),
       UNIQUE KEY uq_box (box),
       KEY idx_bigbox (bigbox),
       KEY idx_pallet (pallet)
       )ENGINE=InnoDB;
